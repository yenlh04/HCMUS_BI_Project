-- Truy vấn tổng số chuyến bay theo tháng, quý, năm. 
SELECT 
    NON EMPTY { [Measures].[Fact Flights Count] } ON COLUMNS, 
    NON EMPTY { 
        (
            [Dim Date].[Year].[Year].ALLMEMBERS * [Dim Date].[Quarter].[Quarter].ALLMEMBERS * [Dim Date].[Month].[Month].ALLMEMBERS 
        ) 
    } DIMENSION PROPERTIES MEMBER_CAPTION, MEMBER_UNIQUE_NAME ON ROWS 
FROM [BI Proj Airlines DDS]
CELL PROPERTIES VALUE, BACK_COLOR, FORE_COLOR, FORMATTED_VALUE, FORMAT_STRING, FONT_NAME, FONT_SIZE, FONT_FLAGS
